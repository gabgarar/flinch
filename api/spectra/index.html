    def extract(path: Path) -> dict[str, np.ndarray]:
    """Compact extract of a level-2 file, as a dictionary of arrays ready for ``numpy.savez``.

    Keys: the header values in ``HEADER_KEYS`` and ``CCF_KEYS`` (lower case); ``activity_names``,
    ``activity_values`` and ``activity_errors`` (the indices computed by the NEID pipeline);
    ``ccf`` (sum over orders of the cross-correlation function, on the velocity grid
    ``ccfstart + ccfstep * i`` km/s); and for each window in ``LINE_WINDOWS`` the arrays
    ``<name>_wave``, ``<name>_flux`` and ``<name>_var``.
    """
    from astropy.io import fits  # optional dependency ("analysis" extra)

    out: dict[str, np.ndarray] = {}
    # Without a memory map: on Windows a mapped file cannot be deleted while an error that
    # happened here is still being reported, and the 113 MB would stay behind.
    with fits.open(path, memmap=False) as hdus:
        if hdus["SCIWAVESTITCH"].data is None:  # some files of the archive come without it
            raise ValueError(f"{Path(path).name} has no stitched spectrum")
        primary = hdus[0].header
        for key in HEADER_KEYS:
            out[key.lower()] = np.float64(primary.get(key, np.nan))
        ccf_header = hdus["CCFS"].header
        for key in CCF_KEYS:
            out[key.lower()] = np.float64(ccf_header.get(key, np.nan))
        out["ccf"] = np.nansum(np.asarray(hdus["CCFS"].data, dtype=float), axis=0)
        activity = hdus["ACTIVITY"].data
        out["activity_names"] = np.array([str(name).strip() for name in activity["INDEX"]])
        out["activity_values"] = np.asarray(activity["VALUE"], dtype=float)
        out["activity_errors"] = np.asarray(activity["UNCERTAINTY"], dtype=float)
        wave = np.asarray(hdus["SCIWAVESTITCH"].data, dtype=float)
        flux = hdus["SCIFLUXSTITCH"].data
        variance = hdus["SCIVARSTITCH"].data
        for name, (low, high) in LINE_WINDOWS.items():
            first, last = np.searchsorted(wave, [low, high])
            out[f"{name}_wave"] = wave[first:last]
            out[f"{name}_flux"] = np.asarray(flux[first:last], dtype=np.float32)
            out[f"{name}_var"] = np.asarray(variance[first:last], dtype=np.float32)
    return out

    def digest(arrays) -> str:
    """Fingerprint of an extract: SHA-256 of everything that is copied from the level-2 file.

    Two extracts of the same file have the same fingerprint on any machine, so it tells whether
    the archive still serves the spectrum that was used. ``ccf`` is left out: it is a sum over
    orders computed here, and its last digits can depend on the numerical library.
    """
    sha = hashlib.sha256()
    for key in sorted(arrays.keys() if hasattr(arrays, "keys") else arrays.files):
        if key == "ccf":
            continue
        value = np.asarray(arrays[key])
        if value.dtype.kind in "US":
            content = "\n".join(str(item) for item in value.ravel()).encode("utf-8")
        else:
            content = np.ascontiguousarray(value.astype(value.dtype.newbyteorder("<"))).tobytes()
        sha.update(f"{key}:{value.dtype.kind}{value.dtype.itemsize}:{value.shape}:".encode("ascii"))
        sha.update(content)
    return sha.hexdigest()

    def fetch_one(filepath: str, filename: str, directory: Path, scratch: Path) -> Path:
    """Download one level-2 file, save its extract in ``directory`` and delete the file."""
    destination = directory / (Path(filename).stem + ".npz")
    if destination.exists():
        return destination
    raw = _retry(lambda: download(f"{FILE_URL}?filehand={filepath}&solar", scratch / filename, overwrite=True), attempts=3)
    try:
        arrays = extract(raw)
        partial = destination.with_name(destination.stem + ".part.npz")  # savez needs the .npz suffix
        np.savez_compressed(partial, **arrays)
        _retry(lambda: partial.replace(destination))
    finally:
        _retry(lambda: raw.unlink(missing_ok=True))
    return destination

    def fetch_window(jd_from: float, jd_to: float, name: str, workers: int = 3, step: int = 1) -> Path:
    """Fetch the extracts of every solar spectrum in a time window; returns their directory.

    Extracts go to ``<data>/spectra/<name>/``. Files already extracted are skipped, so an
    interrupted run can simply be repeated. A file that fails is reported and left for a rerun.
    With ``step`` = n only every n-th spectrum is fetched: a quick first pass at lower cadence,
    which a later run with a smaller step completes.
    """
    directory = data_dir() / "spectra" / name
    directory.mkdir(parents=True, exist_ok=True)
    files = list_files(jd_from, jd_to).iloc[::step]
    scratch = Path(tempfile.gettempdir()) / "flinch_spectra"
    scratch.mkdir(parents=True, exist_ok=True)

    def task(row) -> str | None:
        try:
            fetch_one(row.l2filepath, row.l2filename, directory, scratch)
            return None
        except Exception as exc:  # noqa: BLE001  (reported; the rest of the window goes on)
            return f"{row.l2filename}: {type(exc).__name__}: {exc}"

    with ThreadPoolExecutor(max_workers=workers) as pool:
        failures = [message for message in pool.map(task, files.itertuples()) if message]
    done = len(list(directory.glob("neidL2_*.npz")))
    print(f"{name}: {done} of {len(files)} spectra extracted in {directory}", flush=True)
    for message in failures:
        print(f"  failed  {message}", flush=True)
    return directory

    def load_window(directory: Path) -> tuple[pd.DataFrame, dict[str, np.ndarray]]:
    """Read the extracts of a window.

    Returns a table with one row per spectrum (time, velocity, every activity index and its
    error as ``<index>_err``) and a dictionary with the stacked arrays: ``ccf`` and, for each
    line window, ``<name>_wave``, ``<name>_flux`` and ``<name>_var`` (spectra along the first axis,
    truncated to the shortest window, whose length can differ by a pixel between spectra).
    """
    rows, arrays = [], {}
    for path in sorted(Path(directory).glob("neidL2_*.npz")):
        with np.load(path, allow_pickle=False) as data:
            row = {"file": path.stem, "obsjd": float(data["obsjd"])}
            row["jd"] = row["obsjd"] + 0.5 * float(data["exptime"]) / 86400.0
            row["rv_ms"] = float(data["ccfrvmod"]) * 1000.0
            row["sunagl"] = float(data["sunagl"])
            for index, value, error in zip(data["activity_names"], data["activity_values"], data["activity_errors"]):
                row[str(index)] = float(value)
                row[f"{index}_err"] = float(error)
            rows.append(row)
            for key in data.files:
                if key == "ccf" or key.endswith(("_wave", "_flux", "_var")):
                    arrays.setdefault(key, []).append(np.asarray(data[key], dtype=float))
    stacked = {}
    for key, values in arrays.items():
        length = min(len(v) for v in values)
        stacked[key] = np.vstack([v[:length] for v in values])
    return pd.DataFrame(rows), stacked

    def select_events(min_class: str = "M1.0", max_class: str | None = None) -> pd.DataFrame:
    """Flares observed at the peak or during the rise, strongest first.

    ``max_class`` is exclusive: ``select_events("M5.0", "X1.0")`` gives the flares from M5.0 up
    to, but not including, X1.0.
    """
    observations = neid.read_solar_table()
    jd = np.sort(observations["obsjd"].to_numpy())
    flares = goes.load_flares()
    flares = flares[(flares["jd_peak"] >= jd.min()) & (flares["jd_peak"] <= jd.max())]
    flares = flares[flares["peak_flux"] >= goes.class_to_flux(min_class)]
    if max_class is not None:
        flares = flares[flares["peak_flux"] < goes.class_to_flux(max_class)]
    flares = flares.reset_index(drop=True)
    covered = coverage.add_coverage(flares, jd)
    # Spectra between the start and the peak: a flare peaking just after the end of the
    # observing day still has its impulsive phase on record.
    start = covered["jd_start"].fillna(covered["jd_peak"] - 10.0 * MINUTE)
    covered["n_rise"] = coverage.count_between(jd, start.to_numpy(), covered["jd_peak"].to_numpy())
    covered = covered[covered["at_peak"] | (covered["n_rise"] >= 3)]
    return covered.sort_values("peak_flux", ascending=False, ignore_index=True)

    def clear_sky(
    events: pd.DataFrame,
    steady_obsjd: np.ndarray,
    before_min: float = 26.0,
    window_min: tuple[float, float] = (-8.0, 10.0),
    min_before: int = 12,
    min_window: int = 9,
) -> pd.Series:
    """Flares with a steady, clear sky before their start and around their peak.

    Counts the spectra of ``steady_obsjd`` (see ``quality.stable_obsjd``) from ``before_min`` to
    2 min before the start of the flare, and in ``window_min`` around its peak. The criterion
    uses the signal-to-noise ratio of the spectra alone, never their content, so it can be
    applied before any spectrum is downloaded.
    """
    steady_obsjd = np.asarray(steady_obsjd, dtype=float)
    start = events["jd_start"].fillna(events["jd_peak"] - 10.0 * MINUTE).to_numpy()
    peak = events["jd_peak"].to_numpy()
    before = coverage.count_between(steady_obsjd, start - before_min * MINUTE, start - 2.0 * MINUTE)
    window = coverage.count_between(steady_obsjd, peak + window_min[0] * MINUTE, peak + window_min[1] * MINUTE)
    return pd.Series((before >= min_before) & (window >= min_window), index=events.index)

    def control_events(events: pd.DataFrame, max_shift_days: int = 12, margin_min: float = 60.0) -> pd.DataFrame:
    """One flare-free epoch per flare: the same time of day on the nearest suitable day.

    A day is suitable when no M or X flare peaks within ``margin_min`` minutes of the shifted
    window and the spectrograph took at least 90 % as many spectra in it as in the window of the
    flare itself (which can be cut short by the start or the end of the observing day). The control
    keeps the class and position of its flare (to pair them) and is named after it.
    """
    observations = np.sort(neid.read_solar_table()["obsjd"].to_numpy())
    flares = goes.load_flares()
    major = np.sort(flares.loc[flares["goes_class"].str[0].isin(["M", "X"]), "jd_peak"].to_numpy())
    rows = []
    for row in events.itertuples():
        start = row.jd_start if row.jd_start == row.jd_start else row.jd_peak - 10.0 * MINUTE
        low, high = start - 30.0 * MINUTE, row.jd_peak + 45.0 * MINUTE
        expected = max(10, coverage.count_between(observations, low, high))
        for shift in (s * d for d in range(1, max_shift_days + 1) for s in (1, -1)):
            quiet = coverage.count_between(major, low + shift - margin_min * MINUTE, high + shift + margin_min * MINUTE) == 0
            observed = coverage.count_between(observations, low + shift, high + shift)
            if quiet and observed >= 0.9 * expected:
                control = row._asdict()
                control.pop("Index", None)
                control.update(jd_start=start + shift, jd_peak=row.jd_peak + shift, parent=event_name(row), shift_days=shift)
                rows.append(control)
                break
    return pd.DataFrame(rows)

    def run(
    hours: float = 7.5,
    max_files: int = 600,
    step: int = 3,
    before_min: float = 30.0,
    after_min: float = 45.0,
    workers: int = 3,
    min_class: str = "M1.0",
    core_before_min: float = 10.0,
    core_after_min: float = 15.0,
    kind: str = "flares",
    usable_only: bool = False,
    max_class: str | None = None,
    clear_only: bool = False,
) -> None:
    """Fetch spectrum extracts flare by flare until the time or the file budget runs out.

    Each flare gets the window from ``before_min`` minutes before its start to ``after_min``
    minutes after its peak. Every spectrum is taken in the core of the flare (from
    ``core_before_min`` minutes before the start to ``core_after_min`` after the peak, where
    things change within a minute) and every ``step``-th one outside it. Extracts already on
    disk are skipped, so the campaign can be stopped and run again.

    With ``kind`` = "controls" the same windows are fetched on a flare-free day for every
    flare that already has spectra (see ``control_events``). With ``usable_only`` the controls
    are limited to the flares marked as usable in ``results/spectral_response.csv``.

    ``max_class`` (exclusive) limits the flares from above. With ``clear_only`` only the flares
    that pass ``clear_sky`` are fetched, and of them only the spectra taken through a steady
    sky: the others would be discarded by the analysis anyway, and each costs 113 MB.
    """
    deadline = time.monotonic() + hours * 3600.0
    events = select_events(min_class, max_class)
    steady = quality.stable_obsjd(neid.read_solar_table()) if clear_only else None
    if clear_only and kind == "flares":
        events = events[clear_sky(events, steady)].reset_index(drop=True)
    root = data_dir() / "spectra"
    if kind == "controls":

    def own_indices(arrays: dict) -> pd.DataFrame:
    """Line-core indices measured on the saved windows of the stitched spectra."""
    out = {}
    for name, (window, centre, half, first, second) in OWN_INDICES.items():
        flux = arrays[f"{window}_flux"]
        if len(flux) == 0:  # every spectrum of the window was rejected
            out[name] = np.array([])

    def changes(table: pd.DataFrame, baseline_end_min: float = -2.0, detrend: bool = False) -> pd.DataFrame:
    """Change of each quantity with respect to the spectra taken before the flare start.

    Indices and line-shape parameters are relative changes (parts per thousand or per million);

        if before.sum() >= (4 if detrend else 2):
            if detrend:
                baseline = np.polyval(np.polyfit(time[before], values[before], 1), time)
            else:
                baseline = np.full(len(values), values[before].mean())
            change = (values / baseline - 1.0) * SCALE[unit] if unit in SCALE else values - baseline
            out[column] = change

        else:
            out[column] = np.nan
            out[f"{column}_sigma"] = np.nan
    return out

    def gaussian_residual(parameters: np.ndarray, v: np.ndarray, y: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    """Inverted Gaussian minus the profile, and its derivatives with respect to the parameters.

    The parameters are the centre (km/s), the depth, the standard deviation (km/s) and the level
    of the continuum.
    """
    centre, depth, sigma, level = parameters
    z = (v - centre) / sigma
    bell = np.exp(-0.5 * z * z)
    derivatives = np.column_stack([-depth * bell * z / sigma, -bell, -depth * bell * z * z / sigma, np.ones_like(v)])
    return level - depth * bell - y, derivatives

    def gaussian_parameters(v: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Least-squares parameters of an inverted Gaussian, carried to convergence.

    ``least_squares`` stops when its step is small, not when it is zero. It leaves the depth and
    the width within a few hundredths of a part per million of the minimum, at a point that
    depends on the machine. Gauss-Newton steps with exact derivatives then take the fit to the
    limit of the arithmetic, so that every machine gives the same parameters.
    """
    start = least_squares(lambda p: gaussian_residual(p, v, y)[0], [v[np.argmin(y)], 1.0 - y.min(), 3.5, 1.0]).x
    parameters = start.copy()
    for _ in range(CONVERGENCE_STEPS):
        residual, derivatives = gaussian_residual(parameters, v, y)
        step = np.linalg.lstsq(derivatives, -residual, rcond=None)[0]
        if not np.all(np.isfinite(step)):
            break
        parameters = parameters + step
        if np.all(np.abs(step) <= CONVERGED * np.maximum(1.0, np.abs(parameters))):
            return parameters
    return start  # a profile that is not a line: the steps do not settle, and the first fit stands

    def bisector_span(velocity: np.ndarray, profile: np.ndarray, top=(0.10, 0.40), bottom=(0.60, 0.90)) -> float:
    """Bisector inverse slope in m/s: mean bisector near the top of the line minus near the bottom.

    The levels are fractions of the line depth, measured from the continuum.
    """
    # Work on a finely resampled profile: taking the minimum and the wings from the 0.25 km/s
    # samples makes the span jump by more than 1 m/s whenever the lowest sample changes.
    coarse_min = velocity[int(np.argmin(profile))]
    fine_v = np.arange(coarse_min - 15.0, coarse_min + 15.0, 0.005)
    fine_p = CubicSpline(velocity, profile)(fine_v)
    i_min = int(np.argmin(fine_p))
    depth = 1.0 - fine_p[i_min]
    levels = 1.0 - depth * np.linspace(0.05, 0.95, 91)
    blue_v, blue_p = fine_v[: i_min + 1][::-1], fine_p[: i_min + 1][::-1]  # from the core outwards
    red_v, red_p = fine_v[i_min:], fine_p[i_min:]
    # Keep the monotonic part of each wing, so that np.interp gets increasing abscissae.
    blue_end = int(np.argmax(blue_p >= levels[0])) + 1 if np.any(blue_p >= levels[0]) else len(blue_p)
    red_end = int(np.argmax(red_p >= levels[0])) + 1 if np.any(red_p >= levels[0]) else len(red_p)
    bisector = 0.5 * (
        np.interp(levels, blue_p[:blue_end], blue_v[:blue_end]) + np.interp(levels, red_p[:red_end], red_v[:red_end])
    )
    fraction = (1.0 - levels) / depth
    upper = bisector[(fraction >= top[0]) & (fraction <= top[1])].mean()
    lower = bisector[(fraction >= bottom[0]) & (fraction <= bottom[1])].mean()
    return float((upper - lower) * 1000.0)

    def measure(ccfs: np.ndarray, start: float, step: float) -> pd.DataFrame:
    """Shape parameters of each CCF: centre (m/s), depth, width (km/s), area (km/s) and bisector span (m/s)."""
    velocity = velocity_grid(start, step, ccfs.shape[1])
    rows = []
    for ccf in ccfs:
        profile = normalize(velocity, ccf)
        centre, depth, fwhm = gaussian_fit(velocity, profile)
        near = np.abs(velocity - centre) <= CONTINUUM_KMS[0]
        rows.append(
            {
                "ccf_rv_ms": centre * 1000.0,
                "ccf_depth": depth,
                "ccf_fwhm_kms": fwhm,
                "ccf_area_kms": float(np.sum(1.0 - profile[near]) * step),
                "ccf_bis_ms": bisector_span(velocity, profile),
            }
        )
    return pd.DataFrame(rows)

    def shifted_profiles(ccfs: np.ndarray, start: float, step: float, half_width_kms: float = 15.0):
    """Normalized CCFs resampled on a common grid centred on each line (removes velocity shifts).

    Returns the grid (km/s from the line centre) and the array (spectra, grid points).
    """
    velocity = velocity_grid(start, step, ccfs.shape[1])
    grid = np.arange(-half_width_kms, half_width_kms + 0.5 * step, step)
    out = np.empty((len(ccfs), len(grid)))
    for k, ccf in enumerate(ccfs):
        profile = normalize(velocity, ccf)
        centre, _, _ = gaussian_fit(velocity, profile)
        # A cubic spline: linear interpolation of a line sampled every 0.25 km/s is wrong by up
        # to a few parts in 10^4 at the core, more than the changes looked for.
        out[k] = CubicSpline(velocity, profile)(centre + grid)
    return grid, out
