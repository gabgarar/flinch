    Reproduction¶

    The numbers of the text, the fingerprints of the inputs, the check of the paper against the tables and the command that runs everything.

    flinch.reproduce ¶

    Reproduce the paper from the public archives, in one command, and report what agrees.

    python -m flinch reproduce downloads the data, runs every step of the analysis, compares each new table with the one kept in the repository and checks every number of the paper against the new tables. The steps are the ordinary commands of the package, run one after another; this module only puts them in order, times them and writes the report. The new tables go to their own folder, so the repository is left as it was.

    The spectra are the expensive part: 1296 files of 113 MB that the archive serves whole, 143 GB in all. They are left out by default. --spectra sample fetches a few of them and compares each with the fingerprint of the one used in the paper; --spectra all fetches them all and repeats the spectral analysis.

    class Report:

    def __init__(self, path: Path) -> None:
        self.path = path
        self.lines: list[str] = []

    def say(self, text: str = "") -> None:
        print(text, flush=True)
        self.lines.append(text)
        self.path.write_text("\n".join(self.lines) + "\n", encoding="utf-8")

    def column_units(name: str) -> dict[str, float]:
    """How much "one" of the measured quantity is in each column of a table.

    The response of the spectra is written as a relative change, in parts per thousand or per
    million of each quantity. A difference of 1e-9 in the quantity is 1e-6 or 1e-3 in those
    columns, and that is what the comparison allows there. Every other column is in its own unit.
    """
    if name != "spectral_response.csv":
        return {}
    from .response import QUANTITIES, SCALE

    return {column + suffix: SCALE[unit] for column, _, unit in QUANTITIES if unit in SCALE for suffix in ("", "_err", "_total_err")}

    def compare_tables(reference: Path, new: Path) -> tuple[str, str]:
    """Compare a new table with the reference one.

    Returns the verdict and a detail: "identical" (the same text), "same numbers" (every number
    within ``RELATIVE_TOLERANCE``), "DIFFERENT" (with the columns that differ and by how much),
    or "missing".
    """
    units = column_units(reference.name)
    if not reference.exists() or not new.exists():
        return "missing", "no reference table" if not reference.exists() else "the step did not write it"
    if reference.read_bytes().replace(b"\r\n", b"\n") == new.read_bytes().replace(b"\r\n", b"\n"):
        return "identical", ""
    old, now = pd.read_csv(reference), pd.read_csv(new)
    if list(old.columns) != list(now.columns):
        return "DIFFERENT", "other columns"
    if len(old) != len(now):
        return "DIFFERENT", f"{len(old)} rows in the repository, {len(now)} now"
    differing = []
    for column in old.columns:
        a, b = old[column], now[column]
        if pd.api.types.is_numeric_dtype(a) and pd.api.types.is_numeric_dtype(b):
            x, y = a.to_numpy(dtype=float), b.to_numpy(dtype=float)
            both = np.isfinite(x) & np.isfinite(y)
            if not np.array_equal(np.isfinite(x), np.isfinite(y)):
                differing.append(f"{column} (values missing in one of them)")
            elif both.any():
                gap = np.abs(x[both] - y[both])
                allowed = RELATIVE_TOLERANCE * np.maximum(units.get(column, 1.0), np.abs(x[both]))
                if np.any(gap > allowed):
                    differing.append(f"{column} (by up to {gap.max():.3g})")
        elif not a.astype(str).equals(b.astype(str)):
            differing.append(column)
    if differing:
        return "DIFFERENT", ", ".join(differing)
    return "same numbers", f"differences below {RELATIVE_TOLERANCE:g}"

    def step(report: Report, out: Path, command: str, *paths: str) -> int:
    """Run ``python -m flinch <command> <paths>``, keep its output in a file and report how it went.

    The command writes its tables and figures under ``out``, not in the repository.
    """
    began = time.monotonic()
    log = out / "logs" / (command.replace(" ", "_").replace("--", "") + ".txt")
    env = dict(os.environ, FLINCH_RESULTS=str(out / "results"), FLINCH_FIGURES=str(out / "site_figures"))
    with open(log, "w", encoding="utf-8") as handle:
        done = subprocess.run(
            [sys.executable, "-m", "flinch", *command.split(), *paths], cwd=REPO_ROOT, env=env, stdout=handle, stderr=subprocess.STDOUT
        )
    seconds = time.monotonic() - began
    verdict = "done" if done.returncode == 0 else f"exit code {done.returncode}"
    report.say(f"  {command:28s} {verdict:14s} {seconds:7.0f} s   output in {log}")
    return done.returncode

    def run(spectra: str = "none", sample: int = 10, download: bool = True, out: Path | None = None) -> int:
    """Run everything and write the report; returns 0 when every table and every number agrees.

    Nothing in the repository is touched: the new tables, the figures, the output of every step
    and the report go to ``out`` (``reproduction/`` by default).
    """
    from . import campaign, inputs, verify

    out = Path(out or REPO_ROOT / "reproduction").resolve()
    committed, new, figures = REPO_ROOT / "results", out / "results", out / "figures"
    for folder in (out / "logs", new, figures):
        folder.mkdir(parents=True, exist_ok=True)
    report = Report(out / "report.txt")
    files = len(campaign.paper_files())
    report.say(f"flinch reproduce, {dt.datetime.now():%Y-%m-%d %H:%M}")
    report.say(f"{platform.platform()}, Python {sys.version.split()[0]}, numpy {np.__version__}, pandas {pd.__version__}")
    report.say(f"Data in {data_dir()}")
    report.say(f"Last day of data: {os.environ.get('FLINCH_DATA_END', 'as in the paper')}")
    plan = {
        "none": "not fetched",
        "sample": f"{sample} files, {sample * campaign.FILE_MB / 1024:.1f} GB",
        "all": f"{files} files, {files * campaign.FILE_MB / 1024:.0f} GB",
    }[spectra]
    report.say(f"Spectra: {plan}")
    report.say()
    failures = 0

    report.say("1. Data")
    if download:
        failures += step(report, out, "download") != 0
    try:
        checked = inputs.check()
        for row in checked.itertuples():
            report.say(f"  {row.input:36s} {row.status}")
        if (checked["status"] != "identical").any():
            report.say("  Some inputs are not those of the paper: the archive has changed, and the tables below may differ for that reason.")
    except Exception as exc:  # noqa: BLE001  (no data: the steps below say so one by one)
        report.say(f"  the inputs could not be read ({type(exc).__name__}: {exc})")
        failures += 1
    report.say()

    report.say("2. Velocity analysis")
    for command in VELOCITY_STEPS:
        failures += step(report, out, command) != 0
    report.say()

    report.say("3. Spectra")
    if spectra != "none":
        code = step(report, out, "paper-spectra" + (f" --sample {sample}" if spectra == "sample" else ""))
        check = data_dir() / "spectra" / "check.csv"
        if check.exists():
            counts = pd.read_csv(check)["status"].value_counts()
            report.say("  extracts compared with those of the paper: " + ", ".join(f"{number} {label}" for label, number in counts.items()))
        failures += code != 0
    recomputed = spectra_complete()
    if recomputed:
        failures += step(report, out, "chromosphere") != 0
    else:
        report.say(f"  spectral analysis not repeated: it needs the {files} spectra of the paper (--spectra all)")
        for name in SPECTRAL_TABLES:  # the check of the numbers below reads them
            if (committed / name).exists():
                shutil.copyfile(committed / name, new / name)
    report.say()

    report.say("4. Figures")
    failures += step(report, out, "paper-figures --out", str(figures)) != 0
    report.say()

    report.say("5. New tables, against those of the repository")
    different = 0
    for name in (*VELOCITY_TABLES, *SPECTRAL_TABLES):
        if name in SPECTRAL_TABLES and not recomputed:
            report.say(f"  {name:30s} not recomputed")
            continue
        verdict, detail = compare_tables(committed / name, new / name)
        different += verdict in ("DIFFERENT", "missing")
        report.say(f"  {name:30s} {verdict}" + (f": {detail}" if detail else ""))
    report.say()

    report.say("6. Numbers of the paper, against the new tables")
    table = verify.run(new)
    table.to_csv(new / "paper_check.csv", index=False)
    wrong = table[~table["ok"]]
    for row in wrong.itertuples():
        report.say(f"  NOT IN AGREEMENT  {row.where}, {row.what}: paper {row.paper}, here {row.here}")
    report.say(f"  {len(table)} numbers checked, {len(table) - len(wrong)} in agreement, {len(wrong)} not")
    if not recomputed:
        report.say("  (the numbers of the spectra were checked against the tables of the repository, not against new ones)")
    report.say()

    ok = failures == 0 and different == 0 and wrong.empty
    summary = f"Not fully reproduced: {failures} steps failed, {different} tables differ, {len(wrong)} numbers of the paper not in agreement."
    report.say("Reproduced." if ok else summary)
    report.say(f"Everything is in {out}: report.txt, results/, figures/ and logs/")
    return 0 if ok else 1

    flinch.verify ¶

    Check every number of the paper against the tables in results/.

    The numbers of the paper are written down here, as printed, with the place where they appear. Each is compared with the value in the tables at the precision of the paper: a number passes when the paper gives a correct rounding of it. Run the analysis first and then this check, and the paper is verified against what the code produces today.

    class Checks:
    """A growing list of comparisons between the paper and the tables."""

    def __init__(self) -> None:
        self.rows: list[dict] = []

    def add(self, where: str, what: str, paper, here, decimals: int = 0) -> None:
        """One or several numbers (a value and its error): ``paper`` as printed, ``here`` from the tables."""
        paper = np.atleast_1d(np.asarray(paper, dtype=float))
        here = np.atleast_1d(np.asarray(here, dtype=float))
        ok = bool(np.all(np.isfinite(here)) and np.all(np.abs(here - paper) <= 0.5 * 10.0**-decimals + 1.0e-9))
        self.rows.append(
            {
                "where": where,
                "what": what,
                "paper": " +/- ".join(_text(value, decimals) for value in paper),
                "here": " +/- ".join("missing" if not np.isfinite(value) else _text(value, decimals + 2) for value in here),
                "ok": ok,
            }
        )

    def table(self) -> pd.DataFrame:
        return pd.DataFrame(self.rows)

    def velocity_checks(checks: Checks, directory: Path) -> None:
    """Table 1, the text of Sect. 4.1 and 4.2, and the appendices on robustness and on the early window."""
    table = _read(directory, "velocity_by_instrument.csv")
    if table is None:
        return _missing(checks, "Table 1", "velocity_by_instrument.csv")
    cell = table.set_index(["instrument", "band"])
    for label, band, statistic, *instruments, limit in TABLE_1:
        for name, (flares, value, error) in zip(("NEID", "HARPS-N", "Both"), instruments):
            row = cell.loc[(name, band)]
            checks.add("Table 1", f"{label}: flares, {name}", flares, row["flares"])
            checks.add("Table 1", f"{label}: value, {name}", (value, error), (row[statistic], row[f"{statistic}_err"]), 3)
        both = cell.loc[("Both", band)]
        checks.add("Table 1", f"{label}: 95 % limit", limit, abs(both[statistic]) + 1.645 * both[f"{statistic}_err"], 2)
    both = cell.loc[("Both", "M and X")]
    for name, flares, controls in (("NEID", 0.40, 0.40), ("HARPS-N", 0.40, 0.41)):
        row = cell.loc[(name, "M and X")]
        checks.add("Sect. 4.1", f"scatter of the changes of the flares, {name} (m/s)", flares, row["flare_std"], 2)
        checks.add("Sect. 4.1", f"scatter of the changes of the controls, {name} (m/s)", controls, row["control_std"], 2)
    limit = (abs(both["rotation"]) + 1.645 * both["rotation_err"]) / robust.V_LIMB_FROM_EARTH_MS
    checks.add("Eq. 9", "limit on K dF/F per X1 flare, in units of 1e-5", 3.7, limit * 1.0e5, 1)
    # What that limit is compared with: the brightening of an X1 flare on the scale of GOES-16, and the shift it gives at the limb.

    if variants is None:

    early = _read(directory, "early_window.csv")

    else:

    def ks_p(a, b) -> float:
    """Two-sample Kolmogorov-Smirnov probability that ``a`` and ``b`` come from one distribution."""
    from scipy import stats  # optional dependency ("analysis" extra)

    a, b = pd.Series(a).dropna(), pd.Series(b).dropna()

    Fingerprints of the input data, to tell whether a download holds the data of the paper.

    def harpsn_series() -> dict:
    """The HARPS-N release, a fixed file: its bytes."""
    path = harpsn.series_path()
    sha = hashlib.sha256()
    rows = -1  # the header
    with open(path, "rb") as handle:
        for block in iter(lambda: handle.read(1 << 20), b""):
            sha.update(block)
            rows += block.count(b"\n")
    return {"input": "HARPS-N series", "rows": rows, "sha256": sha.hexdigest()}

    def one_minute_fluxes() -> list[dict]:
    """Each yearly file of one-minute X-ray fluxes that the analysis reads, up to ``DATA_END``.

    They are the years, by satellite, of the M and X flares of the NEID era.
    """
    sample = pilot.major_flare_sample(goes.load_flares(), neid.read_solar_table()["obsjd"].to_numpy())
    needed = set(zip(sample["satellite"].astype(int), jd_to_datetime(sample["jd_peak"]).year))
    out = []
    end = data_end_jd()
    for path in sorted((data_dir() / "goes").glob("sci_xrsf-l2-avg1m_g*_y*_v*.nc")):
        satellite, year = re.search(r"_g(\d\d)_y(\d{4})_", path.name).groups()
        if (int(satellite), int(year)) not in needed:
            continue
        with h5py.File(path, "r") as handle:
            jd = JD_J2000 + handle["time"][:] / 86400.0
            flux = handle["xrsb_flux"][:].astype(float)
        keep = jd < end
        out.append(
            {"input": f"GOES-{satellite} one-minute fluxes of {year}", "rows": int(keep.sum()), "sha256": _sha([_binary(jd[keep]), _binary(flux[keep])])}
        )
    return out

    def check() -> pd.DataFrame:
    """Compare the inputs on disk with those of the paper.

    One row per input of the paper, with ``status``: "identical", "different" or "missing".
    Inputs on disk that the paper did not use (other years of one-minute fluxes) are left out.
    """
    here = fingerprints().set_index("input")
    rows = []
    for row in reference().itertuples():
        if row.input not in here.index:
            status, found = "missing", np.nan
        else:
            found = int(here.at[row.input, "rows"])
            status = "identical" if here.at[row.input, "sha256"] == row.sha256 else "different"
        rows.append({"input": row.input, "rows_paper": row.rows, "rows_here": found, "status": status})
    return pd.DataFrame(rows)

    def freeze(path: Path | None = None) -> Path:
    """Write the fingerprints of the data on disk as the reference. Run by the author when the sample of the paper was closed."""
    path = Path(path or reference_dir() / REFERENCE)
    path.parent.mkdir(parents=True, exist_ok=True)
    fingerprints().to_csv(path, index=False, lineterminator="\n")
    return path
