    def robust_sigma(values: np.ndarray) -> float:
    """Standard deviation estimated from the median absolute deviation."""
    values = np.asarray(values, dtype=float)
    return float(1.4826 * np.median(np.abs(values - np.median(values))))

    def stable_obsjd(table: pd.DataFrame, snr_tolerance: float = 0.01) -> np.ndarray:
    """Exposure start times (sorted) of the spectra taken through a steady, clear sky.

    These are the spectra with a valid velocity error and a signal-to-noise ratio within
    ``snr_tolerance`` of its running median. Unlike ``good`` it does not drop whole days, so it
    suits the spectral indices, which a passing cloud ruins just as it ruins the velocity.

    prepared = table if "snr_rel" in table else prepare(table)
    steady = (prepared["snr_rel"] - 1.0).abs() <= snr_tolerance
    return np.sort(prepared.loc[steady, "obsjd"].to_numpy())

    def matches(obsjd, keep_obsjd, tolerance: float = 2.0e-6) -> np.ndarray:
    """Which exposure start times are in the sorted array ``keep_obsjd``, to 0.2 s by default."""
    obsjd = np.asarray(obsjd, dtype=float)
    keep_obsjd = np.asarray(keep_obsjd, dtype=float)
    if len(keep_obsjd) == 0:
        return np.zeros(len(obsjd), dtype=bool)
    after = np.clip(np.searchsorted(keep_obsjd, obsjd), 0, len(keep_obsjd) - 1)
    before = np.clip(after - 1, 0, len(keep_obsjd) - 1)
    nearest = np.minimum(np.abs(keep_obsjd[after] - obsjd), np.abs(keep_obsjd[before] - obsjd))
    return nearest < tolerance

    def detrend_days(
    jd: np.ndarray,
    rv: np.ndarray,
    days: np.ndarray,
    good: np.ndarray,
    order: int = 2,
    clip: float = 5.0,
    min_points: int = 40,
    max_day_sigma: float = 1.5,
) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Subtract a polynomial from the velocities of each day; the arrays must be sorted in time.

    The polynomial of degree ``order`` is fitted to the good points of the day, dropping those
    beyond ``clip`` robust sigmas until the selection settles. Days left with fewer than
    ``min_points`` good points, or with a robust scatter above ``max_day_sigma``, are dropped
    whole. Returns the residuals (NaN on dropped days), the scatter of each day and the updated
    mask of good points.
    """
    good = np.array(good, dtype=bool)  # writable copy
    residual = np.full(len(jd), np.nan)
    day_sigma = np.full(len(jd), np.nan)
    boundaries = np.flatnonzero(np.diff(days)) + 1
    for lo, hi in zip(np.r_[0, boundaries], np.r_[boundaries, len(days)]):
        mask = good[lo:hi].copy()
        if mask.sum() < min_points:
            good[lo:hi] = False
            continue
        hours = (jd[lo:hi] - jd[lo:hi][mask].mean()) * 24.0
        velocity = rv[lo:hi]
        for _ in range(6):
            fit = np.polyfit(hours[mask], velocity[mask], order)
            r = velocity - np.polyval(fit, hours)
            sigma = robust_sigma(r[mask])
            updated = good[lo:hi] & (np.abs(r) < clip * sigma)
            if updated.sum() < min_points or np.array_equal(updated, mask):
                mask = updated
                break
            mask = updated
        if mask.sum() < min_points:
            good[lo:hi] = False
            continue
        day_sigma[lo:hi] = robust_sigma(r[mask])
        if day_sigma[lo] > max_day_sigma:
            good[lo:hi] = False
            continue
        residual[lo:hi] = r
        good[lo:hi] = mask
    return residual, day_sigma, good

    def prepare(
    table: pd.DataFrame,
    snr_window: int = 31,
    snr_tolerance: float = 0.01,
    max_rv_err_ms: float = 1.0,
    min_snr: float = 50.0,
    order: int = 2,
    clip: float = 5.0,
    min_points: int = 40,
    max_day_sigma: float = 1.5,
) -> pd.DataFrame:
    """Flag good observations and subtract a polynomial trend from each day.

    Added columns:

    ``jd``      mid-exposure Julian date (UTC).
    ``day``     integer UT day number.
    ``snr_rel`` signal-to-noise ratio divided by its running median over ``snr_window`` spectra.
    ``res``     velocity minus the polynomial of degree ``order`` fitted to the good points of the day (m/s).
    ``day_sigma`` robust scatter of the good residuals of the day (m/s).
    ``good``    passes every cut.

    Cuts: a valid velocity error below ``max_rv_err_ms``; a signal-to-noise ratio above
    ``min_snr`` and within ``snr_tolerance`` of its running median (passing clouds lower it and
    bias the velocity); a residual within ``clip`` robust sigmas of the daily fit; at least
    ``min_points`` good spectra in the day; and a daily scatter below ``max_day_sigma`` (the
    median day has 0.8 m/s; days of broken cloud reach tens of m/s and are dropped whole).
    """
    out = table.sort_values("obsjd", ignore_index=True).copy()
    exposure = out["exptime"].where(out["exptime"] > 0, DEFAULT_EXPOSURE_S) if "exptime" in out else DEFAULT_EXPOSURE_S
    out["jd"] = out["obsjd"] + 0.5 * exposure / 86400.0
    out["day"] = np.floor(out["jd"] - 0.5).astype(np.int64)

    valid = (
        np.isfinite(out["rv_ms"])
        & (out["rv_err_ms"] > 0.0)
        & (out["rv_err_ms"] < max_rv_err_ms)
        & (out["extsnr"] > min_snr)
    )
    snr = out["extsnr"].where(valid)
    reference = snr.groupby(out["day"]).transform(
        lambda s: s.rolling(snr_window, center=True, min_periods=max(3, snr_window // 4)).median()
    )
    out["snr_rel"] = snr / reference
    good = np.array(valid & ((out["snr_rel"] - 1.0).abs() <= snr_tolerance), dtype=bool)  # writable copy

    residual, day_sigma, good = detrend_days(
        out["jd"].to_numpy(), out["rv_ms"].to_numpy(), out["day"].to_numpy(), good, order, clip, min_points, max_day_sigma
    )
    out["res"] = residual
    out["day_sigma"] = day_sigma
    out["good"] = good
    return out

    class Series:
    """Time-sorted values with prefix sums, for fast means over many windows."""

    def __init__(self, jd, values):
        jd = np.asarray(jd, dtype=float)
        values = np.asarray(values, dtype=float)
        order = np.argsort(jd, kind="stable")
        self.jd = jd[order]
        self._cumulative = np.r_[0.0, np.cumsum(values[order])]

    def window(self, low, high) -> tuple[np.ndarray, np.ndarray]:
        """Number of points and mean value in each closed interval [low, high] (NaN if empty)."""
        first = np.searchsorted(self.jd, low, side="left")
        last = np.searchsorted(self.jd, high, side="right")
        count = last - first
        total = self._cumulative[last] - self._cumulative[first]
        with np.errstate(invalid="ignore", divide="ignore"):
            mean = np.where(count > 0, total / np.maximum(count, 1), np.nan)
        return count, mean

    def measure(events: pd.DataFrame, series: Series, windows: Windows = Windows()) -> pd.DataFrame:
    """Velocity change of each event (m/s) with respect to its own pre-event baseline.

    ``delta``            mean in the signal window minus mean in the pre-event window.
    ``delta_two_sided``  the same, with the average of the pre and post windows as baseline
                         (cancels a linear trend, but is biased if the response outlasts the peak).
    ``delta_decay``      mean in the decay window minus mean in the pre-event window.
    ``measured``         enough points before and during the event: the minimum number and at
                         least ``min_fraction`` of those expected, so that windows riddled
                         with rejected spectra (passing clouds) are left out.
    """
    start, peak = event_times(events)
    n_pre, pre = series.window(start + windows.pre[0] * MINUTE, start + windows.pre[1] * MINUTE)
    n_signal, signal = series.window(start, peak + windows.signal_after_peak * MINUTE)
    n_post, post = series.window(peak + windows.post[0] * MINUTE, peak + windows.post[1] * MINUTE)
    n_decay, decay = series.window(peak + windows.decay[0] * MINUTE, peak + windows.decay[1] * MINUTE)
    out = events.copy()
    out["rise_min"] = (peak - start) / MINUTE
    out["n_pre"], out["n_signal"], out["n_post"], out["n_decay"] = n_pre, n_signal, n_post, n_decay
    out["baseline"] = pre
    out["delta"] = signal - pre
    out["delta_two_sided"] = signal - 0.5 * (pre + post)
    out["delta_decay"] = decay - pre
    per_minute = windows.min_fraction * 60.0 / windows.cadence_s
    need_pre = np.maximum(windows.min_pre, per_minute * (windows.pre[1] - windows.pre[0]))
    need_signal = np.maximum(windows.min_signal, per_minute * ((peak - start) / MINUTE + windows.signal_after_peak))
    need_post = np.maximum(windows.min_post, per_minute * (windows.post[1] - windows.post[0]))
    out["measured"] = (n_pre >= need_pre) & (n_signal >= need_signal)
    out["measured_two_sided"] = out["measured"] & (n_post >= need_post)
    return out

    def profiles(events: pd.DataFrame, series: Series, edges_min, windows: Windows = Windows()) -> np.ndarray:
    """Mean residual of each event in time bins relative to its peak, minus its pre-event mean.

    Returns an array (events, bins); empty bins are NaN.
    """
    start, peak = event_times(events)
    _, pre = series.window(start + windows.pre[0] * MINUTE, start + windows.pre[1] * MINUTE)
    edges = np.asarray(edges_min, dtype=float)
    out = np.full((len(events), len(edges) - 1), np.nan)
    for k in range(len(edges) - 1):
        # Half-open bins: nudge the upper edge so that a point is never counted twice.
        _, mean = series.window(peak + edges[k] * MINUTE, peak + edges[k + 1] * MINUTE - 1e-9)
        out[:, k] = mean - pre
    return out

    def draw_controls(
    flares: pd.DataFrame,
    major_peaks_jd,
    max_shift_days: int = 15,
    margin_min: float = 45.0,
    seed: int = 0,
) -> pd.DataFrame:
    """Candidate flare-free epochs for each flare, at the same time of day on nearby days.

    Every flare is shifted by whole days (so that the Sun is at the same height and the
    instrument in the same state of its daily cycle) up to ``max_shift_days`` either way. A
    candidate is kept when no major flare peaks between ``margin_min`` minutes before its start
    and ``margin_min`` minutes after its peak. ``rank`` is a random order within each flare.
    """
    rng = np.random.default_rng(seed)
    start, peak = event_times(flares)
    shifts = np.r_[-np.arange(1, max_shift_days + 1), np.arange(1, max_shift_days + 1)].astype(float)
    major = np.sort(np.asarray(major_peaks_jd, dtype=float))
    shifted_start = (start[:, None] + shifts[None, :]).ravel()
    shifted_peak = (peak[:, None] + shifts[None, :]).ravel()
    parent = np.repeat(np.arange(len(flares)), len(shifts))
    low = shifted_start - margin_min * MINUTE
    high = shifted_peak + margin_min * MINUTE
    quiet = np.searchsorted(major, high, side="right") == np.searchsorted(major, low, side="left")
    controls = pd.DataFrame(
        {
            "parent": parent[quiet],
            "shift_days": np.tile(shifts, len(flares))[quiet],
            "jd_start": shifted_start[quiet],
            "jd_peak": shifted_peak[quiet],
            "rank": rng.random(int(quiet.sum())),
        }
    )
    for column in ("goes_class", "peak_flux", "x_norm", "y_norm", "r_norm"):
        if column in flares:
            controls[column] = flares[column].to_numpy()[controls["parent"].to_numpy()]
    return controls

    def measure_power(
    events: pd.DataFrame, squares: Series, differences: Series, windows: Windows = Windows()

    """Run the pilot; returns the tables that the command line prints and writes."""

    def day_bootstrap(values: dict[str, np.ndarray], days: np.ndarray, statistics: dict, draws: int = 4000, seed: int = 0) -> dict:
    """Value and standard error of each statistic, resampling observing days with replacement."""
    rng = np.random.default_rng(seed)
    unique, inverse = np.unique(days, return_inverse=True)
    members = [np.flatnonzero(inverse == k) for k in range(len(unique))]
    out = {name: [function(values)] for name, function in statistics.items()}
    for _ in range(draws):
        rows = np.concatenate([members[k] for k in rng.integers(len(unique), size=len(unique))])
        resampled = {key: array[rows] for key, array in values.items()}
        for name, function in statistics.items():
            out[name].append(function(resampled))
    return {name: (results[0], float(np.nanstd(results[1:], ddof=1))) for name, results in out.items()}

    those that start less than ``ISOLATION_MIN`` minutes after the peak of another M or X
    flare, whose window before the start is not free of flares.
    """
    all_flares = goes.load_flares()
    rows = []
    for instrument in (NEID, HARPSN):
        raw = raw_table(instrument)
        for label, choices in cleaning_variants(instrument):
            prepared = prepare(instrument, raw, **choices)
            good = prepared[prepared["good"]]
            out = measure(instrument, prepared, all_flares, seed)
            subsets = [(label, out["flares"])]
            if not choices:
                alone = not_preceded(out["flares"], out["sample"]["jd_peak"])
                subsets.append((ISOLATED, out["flares"][alone].reset_index(drop=True)))
            for variant, flares in subsets:
                table = band_table(instrument, flares, out["controls"], seed).set_index("band")
                for band in ("M and X", "X"):
                    row = table.loc[band]
                    rows.append(
                        {
                            "instrument": instrument.name,
                            "variant": variant,
                            "band": band,
                            "good": len(good),
                            "days": int(good["day"].nunique()),
                            "flares": int(row["flares"]),
                            "mean": row["mean"],
                            "mean_err": row["mean_err"],
                            "rotation": row["rotation"],
                            "rotation_err": row["rotation_err"],
                        }
                    )
    return pd.DataFrame(rows)

    def run(seed: int = 0) -> dict:
    """The velocity response measured with NEID and with HARPS-N, and both combined."""
    all_flares = goes.load_flares()
    tables, profiles, early, events, numbers = [], [], [], [], {}
    for instrument in (NEID, HARPSN):
        raw = raw_table(instrument)
        prepared = prepare(instrument, raw)
        out = measure(instrument, prepared, all_flares, seed)
        flares, controls = out["flares"], out["controls"]
        tables.append(band_table(instrument, flares, controls, seed))
        profiles.append(stacked_profiles(instrument, flares, controls, out["series"]))
        # Every flare with data in the early window, whether or not its other windows are complete.
        early.append(early_window(instrument, out["sample"], out["series"]))
        events.append(flares.assign(instrument=instrument.name, sample="flare"))
        events.append(controls.assign(instrument=instrument.name, sample="control"))
        good = prepared[prepared["good"]]
        numbers[instrument.name] = {
            "observations": len(prepared),
            "good": len(good),
            "days": int(good["day"].nunique()),
            "point_sigma_ms": quality.robust_sigma(good["res"].to_numpy()),
            "first": float(prepared["obsjd"].min()),
            "last": float(prepared["obsjd"].max()),
            "flares_in_era": len(out["sample"]),
            "flares_measured": len(flares),
            "injection_efficiency": injection_efficiency(instrument, raw, controls, seed=seed),
            "flares_paired": int(np.isfinite(flares["d"]).sum()),
            "controls": len(controls),
        }
    shared = pd.merge_asof(
        events[0][["jd_peak"]].sort_values("jd_peak"), events[2][["jd_peak"]].assign(other=1).sort_values("jd_peak"),
        on="jd_peak", tolerance=1.0 * MINUTE, direction="nearest",
    )
    numbers["flares_in_both"] = int(shared["other"].notna().sum())
    columns = ["instrument", "sample", "jd_start", "jd_peak", "goes_class", "peak_flux", "x_norm", "y_norm", "r_norm", "delta", "d"]
    table = pd.concat([*tables, combine(tables)], ignore_index=True)
    both = table[(table["instrument"] == "Both") & (table["band"] == "M and X")].iloc[0]
    numbers["optical_contrast_limit_per_X1"] = float((abs(both["rotation"]) + 1.645 * both["rotation_err"]) / robust.V_LIMB_FROM_EARTH_MS)
    # A shift common to all flares is measured back as the weighted mean of what each instrument recovers.
    alone = table[(table["band"] == "M and X") & (table["instrument"] != "Both")]
    recovered = robust.recovered_fraction([numbers[name]["injection_efficiency"] for name in alone["instrument"]], alone["mean_err"])
    numbers["detection_probability_10cm"] = robust.detection_probability(recovered * 0.10, float(both["mean_err"]))
    return {
        "table": table,
        "profiles": pd.concat(profiles, ignore_index=True),
        "early_window": pd.concat(early, ignore_index=True),
        "events": pd.concat(events, ignore_index=True).reindex(columns=columns),
        "numbers": numbers,
    }
